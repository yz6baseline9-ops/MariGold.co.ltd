        function setMental(status) {

            currentMental = status;

                btn.className = "mental-btn p-3 border-2 border-slate-200 rounded-xl text-center bg-white hover:border-slate-400 transition flex flex-col items-center justify-center";

                textSpan.className = "text-xs font-bold text-slate-600";

            });

                activeBtn.className = "mental-btn p-3 border-2 border-emerald-500 rounded-xl text-center bg-emerald-50/60 transition flex flex-col items-center justify-center shadow-xs";
